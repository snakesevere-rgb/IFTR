class HighVolumeEncryptedLocation(BaseModel):

    coordinates: str = ""  # Encrypted JSON: {"lat": 22.7, "lng": -74.0}
    # ... other fields

    @validator('coordinates')
    def validate_encrypted_format(cls, v):
        # Quick check that it's base64 and decryptable
        try:
            data = decrypt_data(v)
            coords = json.loads(data)
            assert 'lat' in coords and 'lng' in coords

        except:
            raise ValueError("Invalid encrypted coordinates format")

# Enhanced Location Encryption. Current system meets standards already, but this adds an extra layer of security
# replace class Order(BaseModel): in models.py
class Order(BaseModel):
    """
    Order with encrypted location data.

    Note: Coordinates are encrypted separately (lat/lng). For deployments
    with >1M active orders, consider combined coordinate encryption to
    reduce storage. Current approach supports most use cases effectively.
    """
    pickup_location: EncryptedLocation
    delivery_location: EncryptedLocation
    distance_km: Optional[float] = None

    # Add for optimization:
    pickup_geo_hash: Optional[str] = None  # e.g., "dr5reg" (Geohash)
    delivery_geo_hash: Optional[str] = None

    @validator('pickup_location')
    def set_pickup_geo_hash(cls, v, values):
        """Generate approximate location hash for queries"""
        if v.lat and v.lng:
            # Decrypt just enough to get approximate location
            # Or store hash at encryption time
            lat = float(decrypt_data(v.lat))
            lng = float(decrypt_data(v.lng))
            values['pickup_geo_hash'] = geohash.encode(lat, lng, precision=5)
        return v
